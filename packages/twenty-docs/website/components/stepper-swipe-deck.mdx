{/* Generated by scripts/docs/render-api-reference.mjs from component-docs.json. Do not edit by hand. */}

<ParamField body="activeIndex" type="number" required>
  The step shown, counted from 0. The deck slides to it.
</ParamField>

<ParamField body="onActiveIndexChange" type="(nextIndex: number) =&gt; void" required>
  Called with the next or previous index when a swipe passes the commit threshold. The parent owns `activeIndex`.
</ParamField>

<ParamField body="stepCount" type="number" required>
  How many slides the deck holds. Dragging past the first or last one meets resistance.
</ParamField>
