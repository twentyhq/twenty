<Check>The popup renders at the end of the page, so pass the surface's `scheme`(Select.tsx:85-86,99-109,323)</Check>

<Check>Values are plain strings; an empty string means nothing selected (Select.tsx:197-198,220)</Check>

## Props

{/* Generated by scripts/docs/render-api-reference.mjs from component-docs.json. Do not edit by hand. */}

<ParamField body="ariaLabel" type="string" required>
  Names both the trigger and the listbox for screen readers.
</ParamField>

<ParamField body="onValueChange" type="(value: TValue) =&gt; void" required>
  Called with an option's value when it is picked. The parent owns `value`.
</ParamField>

<ParamField body="options" type="readonly SelectOption&lt;TValue&gt;[]" required>
  The choices, in order: a `label` and a `value` each.
</ParamField>

<ParamField body="placeholder" type="string" required>
  Shown in the trigger while nothing is picked.
</ParamField>

<ParamField body="value" type="string" required>
  The picked value, or an empty string for none.
</ParamField>

<ParamField body="emptyLabel" type="string">
  Shown in the popup when no option matches. Defaults to a dash.
</ParamField>

<ParamField body="invalid" type="boolean" default="false">
  Paints the trigger's red border.
</ParamField>

<ParamField body="scheme" type="&quot;light&quot; | &quot;muted&quot; | &quot;dark&quot;" default="light">
  The popup's scheme. It renders in a portal outside the section, so it cannot inherit the section's scheme and takes it here.
</ParamField>

<ParamField body="searchable" type="boolean" default="false">
  Adds a filter row at the top of the popup, matching the option labels.
</ParamField>

<ParamField body="searchPlaceholder" type="string">
  Shown in the filter row while it is empty.
</ParamField>
