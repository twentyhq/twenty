## Usage guidelines

<Check>Tags are trimmed; blanks and exact duplicates are ignored (add-tag.ts:1-6)</Check>

<Check>The draft is added on blur; Backspace on an empty draft removes the last tag (TagInput.tsx:214-216,242)</Check>

## Props

{/* Generated by scripts/docs/render-api-reference.mjs from component-docs.json. Do not edit by hand. */}

<ParamField body="ariaLabel" type="string" required>
  The text input's accessible name.
</ParamField>

<ParamField body="onValuesChange" type="(values: string[]) =&gt; void" required>
  Called with the new list when a tag is added (Enter or comma) or removed (its button, or Backspace in an empty input). The parent owns `values`.
</ParamField>

<ParamField body="removeLabel" type="(tag: string) =&gt; string" required>
  Builds each tag's remove-button label from the tag, so it can be translated.
</ParamField>

<ParamField body="suggestions" type="readonly string[]" required>
  Values offered up front, minus those already chosen. The typing menu matches them too.
</ParamField>

<ParamField body="values" type="readonly string[]" required>
  The tags chosen, in order.
</ParamField>

<ParamField body="placeholder" type="string">
  Shown in the input only while no tag is chosen.
</ParamField>

<ParamField body="searchPool" type="readonly string[]">
  Extra values the typing menu can match, beyond `suggestions`, without showing them up front.
</ParamField>
