## Props

{/* Generated by scripts/docs/render-api-reference.mjs from component-docs.json. Do not edit by hand. */}

<ParamField body="ariaLabel" type="string" required>
  Names the group of pills for screen readers (`role="group"`).
</ParamField>

<ParamField body="onToggle" type="(value: TValue) =&gt; void" required>
  Called with a pill's value when it is clicked. The parent adds or removes it from `values`.
</ParamField>

<ParamField body="options" type="readonly ChipOption&lt;TValue&gt;[]" required>
  The pills, in order: a `label` and a `value` each.
</ParamField>

<ParamField body="values" type="readonly TValue[]" required>
  The selected values. Each selected pill is pressed (`aria-pressed`).
</ParamField>

<ParamField body="invalid" type="boolean" default="false">
  Sets `aria-invalid` on the group and paints the unselected pills red, so the missing choice shows.
</ParamField>
